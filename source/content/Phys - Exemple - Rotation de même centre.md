\begin{align*}
\overrightarrow{e_{x}}&=\cos\theta \overrightarrow{i}+\sin\theta \overrightarrow{j}&&\cr
\dot{\overrightarrow{e_{x}}}&=\dot{\theta}\sin\theta \overrightarrow{i}-\dot{\theta}\cos\theta \overrightarrow{j}&&\cr
&=\dot{\theta}\overrightarrow{y}&&\cr
\end{align*}
$$

$$
\dot{\overrightarrow{e_{c}}}=\overrightarrow{\Omega}\wedge \overrightarrow{e_{c}}.
